Existing template to copy verbatim (from data/SettingsRepository.kt):
```kotlin
interface SettingsRepository {
    val hintsEnabled: Flow<Boolean>
    suspend fun setHintsEnabled(enabled: Boolean)
}
internal object SettingsPreferences {
    val HINTS_KEY = booleanPreferencesKey("hints_enabled")
    const val DEFAULT_HINTS = true
}
class SettingsRepositoryImpl @Inject constructor(private val dataStore: DataStore<Preferences>) : SettingsRepository {
    override val hintsEnabled: Flow<Boolean> =
        dataStore.data.catch { emit(emptyPreferences()) }.map { it[SettingsPreferences.HINTS_KEY] ?: SettingsPreferences.DEFAULT_HINTS }
    override suspend fun setHintsEnabled(enabled: Boolean) { dataStore.edit { it[SettingsPreferences.HINTS_KEY] = enabled } }
}
```
